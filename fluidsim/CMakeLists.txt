cmake_minimum_required(VERSION 3.16)
project(fluidsim CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Solver choice (also selectable by editing src/config.h):  -DFLUID_SOLVER=SPH | EULER
set(FLUID_SOLVER "" CACHE STRING "SPH or EULER (empty = use src/config.h)")

find_package(Vulkan REQUIRED)
find_package(glfw3 3.3 REQUIRED)
find_package(OpenMP)

add_executable(fluidsim src/main.cpp src/vk_renderer.cpp)
target_link_libraries(fluidsim PRIVATE Vulkan::Vulkan glfw)
if(OpenMP_CXX_FOUND)
  target_link_libraries(fluidsim PRIVATE OpenMP::OpenMP_CXX)
endif()
if(FLUID_SOLVER STREQUAL "SPH")
  target_compile_definitions(fluidsim PRIVATE FLUID_SOLVER=0)
elseif(FLUID_SOLVER STREQUAL "EULER")
  target_compile_definitions(fluidsim PRIVATE FLUID_SOLVER=1)
endif()

# ---- shaders: GLSL -> SPIR-V (glslc from the Vulkan SDK, or glslangValidator)
find_program(GLSLC glslc HINTS $ENV{VULKAN_SDK}/bin $ENV{VULKAN_SDK}/Bin)
find_program(GLSLANG glslangValidator HINTS $ENV{VULKAN_SDK}/bin $ENV{VULKAN_SDK}/Bin)
set(SHADER_OUT ${CMAKE_BINARY_DIR}/shaders)
file(MAKE_DIRECTORY ${SHADER_OUT})
set(SPV)
foreach(stage vert frag)
  set(src ${CMAKE_SOURCE_DIR}/shaders/particle.${stage})
  set(spv ${SHADER_OUT}/particle.${stage}.spv)
  if(GLSLC)
    add_custom_command(OUTPUT ${spv} COMMAND ${GLSLC} ${src} -o ${spv} DEPENDS ${src})
  elseif(GLSLANG)
    add_custom_command(OUTPUT ${spv} COMMAND ${GLSLANG} -V ${src} -o ${spv} DEPENDS ${src})
  else()
    message(FATAL_ERROR "Need glslc or glslangValidator (install the Vulkan SDK)")
  endif()
  list(APPEND SPV ${spv})
endforeach()
add_custom_target(shaders DEPENDS ${SPV})
add_dependencies(fluidsim shaders)
target_compile_definitions(fluidsim PRIVATE SHADER_DIR="${SHADER_OUT}")
